cmake_minimum_required(VERSION 3.16)
project(minivectordb_lite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(minivectordb_lite src/minivectordb_lite.cpp)
target_compile_options(minivectordb_lite PRIVATE -O3 -DNDEBUG -ffast-math -march=native -Wall -Wextra)
