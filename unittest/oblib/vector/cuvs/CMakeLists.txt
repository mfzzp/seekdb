cmake_minimum_required(VERSION 3.16)
project(cuvs_host_checks LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
get_filename_component(SEEKDB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)

# These doubles test bridge ownership, stream ordering and error handling only.
# They are never included or linked by the production bridge build.
add_library(bridge_under_test OBJECT "${SEEKDB_ROOT}/tools/hipvs_bridge/seekdb_cuvs_bridge.c")
target_include_directories(bridge_under_test PRIVATE stubs)
add_executable(test_cuvs_bridge test_cuvs_bridge.cpp $<TARGET_OBJECTS:bridge_under_test>)
target_include_directories(test_cuvs_bridge PRIVATE stubs "${SEEKDB_ROOT}/tools/hipvs_bridge")
add_test(NAME cuvs_bridge_lifecycle COMMAND test_cuvs_bridge)

add_executable(test_cuvs_worker test_cuvs_worker.cpp)
target_include_directories(test_cuvs_worker PRIVATE "${SEEKDB_ROOT}/src/oblib")
if(WIN32)
  if(NOT PTHREADS_ROOT)
    message(FATAL_ERROR "Windows tests require PTHREADS_ROOT (pthreads4w include/lib/bin)")
  endif()
  target_include_directories(test_cuvs_worker PRIVATE "${PTHREADS_ROOT}/include")
  target_link_libraries(test_cuvs_worker PRIVATE "${PTHREADS_ROOT}/lib/pthreadVC3.lib")
  add_custom_command(TARGET test_cuvs_worker POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${PTHREADS_ROOT}/bin/pthreadVC3.dll" "$<TARGET_FILE_DIR:test_cuvs_worker>")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(test_cuvs_worker PRIVATE Threads::Threads)
endif()
add_test(NAME cuvs_worker_lifecycle COMMAND test_cuvs_worker)
add_executable(test_cuvs_row_batch test_cuvs_row_batch.cpp)
target_include_directories(test_cuvs_row_batch PRIVATE "${SEEKDB_ROOT}/src/oblib")
add_test(NAME cuvs_row_batch COMMAND test_cuvs_row_batch)
set_tests_properties(cuvs_bridge_lifecycle cuvs_worker_lifecycle cuvs_row_batch PROPERTIES TIMEOUT 60)

add_executable(test_cuvs_snapshot_io test_cuvs_snapshot_io.cpp)
target_include_directories(test_cuvs_snapshot_io PRIVATE "${SEEKDB_ROOT}/src/oblib")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(test_cuvs_snapshot_io PRIVATE "-Wl,--wrap=mmap" "-Wl,--wrap=munmap")
endif()
add_test(NAME cuvs_snapshot_io COMMAND test_cuvs_snapshot_io)
set_tests_properties(cuvs_snapshot_io PROPERTIES TIMEOUT 60)

add_executable(test_cuvs_snapshot_format test_cuvs_snapshot_format.cpp)
target_include_directories(test_cuvs_snapshot_format PRIVATE "${SEEKDB_ROOT}/src/oblib")
add_test(NAME cuvs_snapshot_format COMMAND test_cuvs_snapshot_format)
set_tests_properties(cuvs_snapshot_format PROPERTIES TIMEOUT 60)

# Compile the same bundled Zstd implementation used by seekdb, without system
# packages, network downloads or substitutes for the compression algorithm.
set(ZSTD_ROOT "${SEEKDB_ROOT}/src/oblib/lib/compress/zstd_1_3_8")
file(GLOB ZSTD_SOURCES "${ZSTD_ROOT}/zstd_src/*.c")
add_library(cuvs_test_zstd STATIC ${ZSTD_SOURCES} "${ZSTD_ROOT}/ob_zstd_wrapper.cpp")
target_compile_definitions(cuvs_test_zstd PRIVATE ZSTDERRORLIB_VISIBILITY= ZDICTLIB_VISIBILITY= ZSTDLIB_VISIBILITY=)
if(NOT WIN32)
  target_compile_options(cuvs_test_zstd PRIVATE -fvisibility=hidden)
endif()
if(NOT MSVC)
  target_compile_options(cuvs_test_zstd PRIVATE -w)
endif()
add_executable(test_cuvs_snapshot_codec test_cuvs_snapshot_codec.cpp)
target_include_directories(test_cuvs_snapshot_codec PRIVATE "${SEEKDB_ROOT}/src/oblib")
target_link_libraries(test_cuvs_snapshot_codec PRIVATE cuvs_test_zstd)
add_test(NAME cuvs_snapshot_codec COMMAND test_cuvs_snapshot_codec)
set_tests_properties(cuvs_snapshot_codec PROPERTIES TIMEOUT 60)
